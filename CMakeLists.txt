cmake_minimum_required(VERSION 3.20)
project(replicated_key_value_store LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(kv src/file.cpp src/wal.cpp src/store.cpp src/net.cpp src/protocol.cpp)
target_include_directories(kv PUBLIC src)
target_include_directories(kv SYSTEM PUBLIC third_party)
target_link_libraries(kv PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(kv PUBLIC ws2_32)
endif()
if(MSVC)
    target_compile_options(kv PRIVATE /W4)
else()
    target_compile_options(kv PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(kv_leader src/leader.cpp)
target_link_libraries(kv_leader PRIVATE kv)
add_executable(kv_replica src/replica.cpp)
target_link_libraries(kv_replica PRIVATE kv)
add_executable(kv_client src/client.cpp)
target_link_libraries(kv_client PRIVATE kv)

include(CTest)
if(BUILD_TESTING)
    add_executable(kv_tests tests/tests.cpp)
    target_link_libraries(kv_tests PRIVATE kv)
    add_test(NAME unit COMMAND kv_tests)
    set_tests_properties(unit PROPERTIES TIMEOUT 25)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
            --leader $<TARGET_FILE:kv_leader> --replica $<TARGET_FILE:kv_replica>
            --client $<TARGET_FILE:kv_client>)
        set_tests_properties(integration PROPERTIES TIMEOUT 120)
    else()
        message(WARNING "Python3 nije pronadjen; integracione testove pokreni rucno")
    endif()
endif()
